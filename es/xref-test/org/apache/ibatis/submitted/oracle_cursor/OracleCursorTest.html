110   private void doTest(Function<Mapper, List<Author>> query, Class<?> expectedBooksType) {
111     try (SqlSession sqlSession = sqlSessionFactory.openSession()) {
112       Mapper mapper = sqlSession.getMapper(Mapper.class);
113       List<Author> authors = query.apply(mapper);
114       assertIterableEquals(
115           List.of(new Author(1, "John", List.of(new Book(1, "C#"), new Book(2, "Python"), new Book(5, "Ruby"))),
116               new Author(2, "Jane", List.of(new Book(3, "SQL"), new Book(4, "Java")))),
117           authors);
118       assertSame(expectedBooksType, authors.get(0).getBooks().getClass());
119     }
120   }

122   @Test

125       Mapper mapper = sqlSession.getMapper(Mapper.class);
126       List<Author2> authors = mapper.selectNestedCursorOfStrings();
127       assertIterableEquals(List.of(new Author2(1, "John", List.of("C#", "Python", "Ruby")),
128           new Author2(2, "Jane", List.of("SQL", "Java"))), authors);
129     }

142   private void doTestBook2(Function<Mapper, List<Book2>> query) {
143     try (SqlSession sqlSession = sqlSessionFactory.openSession()) {
144       Mapper mapper = sqlSession.getMapper(Mapper.class);
145       List<Book2> books = query.apply(mapper);
146       assertIterableEquals(List.of(new Book2(1, "C#", new Author(1, "John")),
147           new Book2(2, "Python", new Author(1, "John")), new Book2(3, "SQL", new Author(2, "Jane")),
148           new Book2(4, "Java", new Author(2, "Jane")), new Book2(5, "Ruby", new Author(1, "John"))), books);
149     }
