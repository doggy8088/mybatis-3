171     try (SqlSession sqlSession = sqlSessionFactory.openSession()) {

173       List<Store10> stores10 = mapper.getStores10();

203   @Test
204   void testImmutableNestedObjectsWithBadEquals() {
205     try (SqlSession sqlSession = sqlSessionFactory.openSession()) {
206       Mapper mapper = sqlSession.getMapper(Mapper.class);
207       List<Container1> containers = mapper.getContainers();
208 
209       Container1 expectedContainer1 = new Container1();
210       expectedContainer1.setNum(1);
211       expectedContainer1.setType("storesWithClerks");
212       expectedContainer1.setStores(Arrays.asList(
213           new Store9(1, "Store 1",
214               Arrays.asList(new Clerk(1001, "Clerk 1001"), new Clerk(1003, "Clerk 1003"),
215                   new Clerk(1004, "Clerk 1004"))),
216           new Store9(2, "Store 2", Arrays.asList()), new Store9(3, "Store 3", Arrays.asList())));
217 
218       Container1 expectedContainer2 = new Container1();
219       expectedContainer2.setNum(1);
220       expectedContainer2.setType("storesWithManagers");
221       expectedContainer2.setStores(Arrays.asList(
222           new Store9(1, "Store 1", Arrays.asList(new Clerk(1002, "Clerk 1002"), new Clerk(1005, "Clerk 1005")))));
223 
224       // cannot use direct equals as we overwrote it with a bad impl on purpose
225       assertThat(containers).isNotNull().hasSize(2);
226       assertContainer1(containers.get(0), expectedContainer1);
227       assertContainer1(containers.get(1), expectedContainer2);
228     }
229   }
230 
231   private static void assertContainer1(Container1 container1, Container1 expectedContainer1) {
232     assertThat(container1).isNotNull().satisfies(c -> {
233       assertThat(c.getNum()).isEqualTo(expectedContainer1.getNum());
234       assertThat(c.getType()).isEqualTo(expectedContainer1.getType());
235       assertThat(c.getStores()).isEqualTo(expectedContainer1.getStores());
236     });
